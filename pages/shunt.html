Shunt Physiology

An interactive model of how pulmonary shunt, tissue oxygen extraction and haemoglobin affinity shape arterial oxygenation and blood oxygen content.

Published

August 23, 2026

How to Use

  1. Set the alveolar oxygen tension (PAO₂).
    Blood leaving ventilated alveoli is assumed to equilibrate completely, so end-capillary PO₂ equals PAO₂.
  2. Adjust the cardiovascular parameters:
    • Cardiac output determines total oxygen delivery. With consumption unchanged, a lower cardiac output forces the tissues to extract more oxygen.
    • Shunt fraction is the proportion of cardiac output that reaches the systemic circulation without taking part in gas exchange, and so retains mixed-venous oxygen content. Increasing the shunt fraction will lower arterial oxygen content.
  3. Set the metabolic demand with the oxygen consumption (VO₂) slider.
  4. Change the haemoglobin concentration to alter oxygen content without changing the saturation associated with a given PO₂.
  5. If desired, adjust the factors that alter the p50 of the oxyhaemoglobin dissociation curve.

Maximum VO2 varies depending on the DO2.

Limitations

This model makes several assumptions:

  • Oxygen transport is at steady state
    i.e. whole-body oxygen uptake equals whole-body oxygen consumption.
  • The Riley 3-compartment model of the lung is used
  • Haemoglobin concentration, cardiac output, oxygen consumption, and shunt fraction are constant across the model
  • PCO₂, plasma pH, 2,3-DPG, and temperature are adjustable but remain constant within each modelled state
    i.e. pH is independent of CO₂
  • Haemoglobinopathies are not modelled

## Validate that a value is a valid numeric variable and sits within the range
validateNumeric = function(value, name, lower = -Inf, upper = Inf) {
    if (!is.numeric(value) || length(value) == 0 || any(!is.finite(value))) {
        stop(name, " must contain finite numbers", call. = FALSE)
    }

    if (any(value < lower) || any(value > upper)) {
        stop(
            name,
            " must be between ",
            lower,
            " and ",
            upper,
            call. = FALSE
        )
    }

    invisible(value)
}

## Validate a value is a single valid numeric
validateScalar = function(value, name, lower = -Inf, upper = Inf) {
    if (!is.numeric(value) || length(value) != 1 || !is.finite(value)) {
        stop(name, " must be one number", call. = FALSE)
    }

    validateNumeric(value, name, lower, upper)
}

## Calculate oxygen content of blood
calcOxygenContent = function(
    po2_mmhg,
    hb_g_l,
    pco2_mmhg = STANDARD_PCO2_MMHG,
    plasma_ph = 7.40,
    dpg_mmol_l = 4.65,
    temperature_c = STANDARD_TEMPERATURE_C
) {
    validateNumeric(po2_mmhg, "po2_mmhg", 0, 1000)
    validateScalar(hb_g_l, "hb_g_l", 0, 300)

    # Get the SpO2, based on the PO2 and the shape of the curve
    spO2 = calculateSpO2(
        po2_mmhg = po2_mmhg,
        pco2_mmhg = pco2_mmhg,
        plasma_ph = plasma_ph,
        dpg_mmol_l = dpg_mmol_l,
        temperature_c = temperature_c
    )

    # Calculate bound and dissolved O2
    ## NB: Divide by 10 as converting from hb_g_**l** to ml_**dl**
    bound_ml_dl = OXYGEN_CAPACITY_ML_G * hb_g_l / 10 * spO2
    dissolved_ml_dl = OXYGEN_SOLUBILITY_ML_DL_MMHG * po2_mmhg

    data.frame(
        po2_mmhg = po2_mmhg,
        saturation = spO2,
        bound_ml_dl = bound_ml_dl,
        dissolved_ml_dl = dissolved_ml_dl,
        total_ml_dl = bound_ml_dl + dissolved_ml_dl
    )
}

## Calculate partial pressure, given a given oxygen content and p50
## Needed for calculating MVO2
calcPO2fromContent = function(
    content_ml_dl,
    hb_g_l,
    pco2_mmhg = STANDARD_PCO2_MMHG,
    plasma_ph = 7.40,
    dpg_mmol_l = 4.65,
    temperature_c = STANDARD_TEMPERATURE_C,
    po2_bounds = c(0, 1000),
    tolerance = 1e-8
) {
    validateScalar(content_ml_dl, "content_ml_dl", 0, Inf)
    validateScalar(hb_g_l, "hb_g_l", 0, 300)
    validateNumeric(po2_bounds, "po2_bounds", 0, 1000)
    validateScalar(tolerance, "tolerance", .Machine$double.eps, 1)

    # Find the PO2 that gives the current oxygen content...
    # ...by walking PO2 up within the bounds until the...
    # ... oxygen content at that PO2 = the measured PO2 (i.e., their difference is 0)
    uniroot(
        f = function(po2_mmhg) {
            calcOxygenContent(
                po2_mmhg = po2_mmhg,
                hb_g_l = hb_g_l,
                pco2_mmhg = pco2_mmhg,
                plasma_ph = plasma_ph,
                dpg_mmol_l = dpg_mmol_l,
                temperature_c = temperature_c
            )$total_ml_dl -
                content_ml_dl
        },
        interval = po2_bounds,
        tol = tolerance
    )$root
}

# Blood and oxygen
## Oxygen content is mL/dL, so the flow is converted to dL/min to match it
calcPulmonaryFlow = function(cardiac_output_l_min, shunt_fraction) {
    validateScalar(cardiac_output_l_min, "cardiac_output_l_min", 2, 12)
    validateScalar(shunt_fraction, "shunt_fraction", 0, 0.6)

    10 * cardiac_output_l_min * (1 - shunt_fraction)
}

## Calculate mixed-venous content from the oxygen the tissues consume
# (At steady state, venous blood entering the pulmonary capillary will take up what was consumed)
# This is done this way to escape the circular argument between Pv, Pa, and PcO2
# Clinically we can measure PaO2 and MVO2, and assume PAO2, but not measure PcO2 (mostly)...
# ... and calculate a shunt fraction
# In this sim we set PAO2 (PcO2), shunt fraction, and VO2
# Therefore MV content *or* arterial content has to be derived algebraically
# Assuming steady-state; 'pure shunt', fixed Co, VO2, etc, etc
# Whole body Fick: VO2 = 10 × Qt × (CaO2 - CvO2)
# and
# Shunt equation: CaO2 = (1 - Qs/Qt) × CcO2 + (Qs/Qt) × CvO2
# Rearranging
# Therefore: CvO2 = CcO2 - VO2 / (10 × Qt × (1 - Qs/Qt))

calcMixedVenousContent = function(
    end_capillary_content_ml_dl,
    vo2_ml_min,
    shunt_fraction,
    cardiac_output_l_min
) {
    validateScalar(
        end_capillary_content_ml_dl,
        "end_capillary_content_ml_dl",
        0,
        Inf
    )
    validateScalar(vo2_ml_min, "vo2_ml_min", 0, Inf)

    pulmonary_flow_dl_min = calcPulmonaryFlow(
        cardiac_output_l_min = cardiac_output_l_min,
        shunt_fraction = shunt_fraction
    )
    mixed_venous_content_ml_dl = end_capillary_content_ml_dl -
        vo2_ml_min / pulmonary_flow_dl_min

    # Consuming more than the circulation delivers would leave the venous...
    # ...blood with negative oxygen...
    if (mixed_venous_content_ml_dl < 0) {
        stop("vo2_ml_min exceeds oxygen delivery", call. = FALSE)
    }

    mixed_venous_content_ml_dl
}

# Calculate arterial content using mixed venous content, end-capillary content, and Qs/Qt
calcArterialContent = function(
    end_capillary_content_ml_dl,
    mixed_venous_content_ml_dl,
    shunt_fraction
) {
    validateScalar(
        end_capillary_content_ml_dl,
        "end_capillary_content_ml_dl",
        0,
        Inf
    )
    validateScalar(
        mixed_venous_content_ml_dl,
        "mixed_venous_content_ml_dl",
        0,
        Inf
    )
    validateScalar(shunt_fraction, "shunt_fraction", 0, 0.6)

    end_capillary_content_ml_dl -
        shunt_fraction *
            (end_capillary_content_ml_dl - mixed_venous_content_ml_dl)
}

## Calculate the proportion of arterial oxygen the tissues remove
calcExtractionRatio = function(
    arterial_content_ml_dl,
    mixed_venous_content_ml_dl
) {
    validateScalar(arterial_content_ml_dl, "arterial_content_ml_dl", 0, Inf)
    validateScalar(
        mixed_venous_content_ml_dl,
        "mixed_venous_content_ml_dl",
        0,
        Inf
    )

    (arterial_content_ml_dl - mixed_venous_content_ml_dl) /
        arterial_content_ml_dl
}

## Calculate the highest oxygen consumption the circululation could theoretically support...
# ... i.e. all o2 consumed
calcMaxVO2 = function(
    end_capillary_content_ml_dl,
    shunt_fraction,
    cardiac_output_l_min
) {
    validateScalar(
        end_capillary_content_ml_dl,
        "end_capillary_content_ml_dl",
        0,
        Inf
    )

    pulmonary_flow_dl_min = calcPulmonaryFlow(
        cardiac_output_l_min = cardiac_output_l_min,
        shunt_fraction = shunt_fraction
    )

    pulmonary_flow_dl_min * end_capillary_content_ml_dl
}

## Calculate the end state produced by shunt and oxygen consumption
calcShuntState = function(
    end_capillary_po2_mmhg,
    shunt_fraction,
    vo2_ml_min,
    cardiac_output_l_min,
    hb_g_l,
    pco2_mmhg = STANDARD_PCO2_MMHG,
    plasma_ph = 7.40,
    dpg_mmol_l = 4.65,
    temperature_c = STANDARD_TEMPERATURE_C
) {
    validateScalar(
        end_capillary_po2_mmhg,
        "end_capillary_po2_mmhg",
        40,
        650
    )
    validateScalar(shunt_fraction, "shunt_fraction", 0, 0.6)
    validateScalar(hb_g_l, "hb_g_l", 50, 200)

    # Start with blood leaving the ventilated alveoli
    end_capillary = calcOxygenContent(
        po2_mmhg = end_capillary_po2_mmhg,
        hb_g_l = hb_g_l,
        pco2_mmhg = pco2_mmhg,
        plasma_ph = plasma_ph,
        dpg_mmol_l = dpg_mmol_l,
        temperature_c = temperature_c
    )

    # The tissues take the whole VO2 out of the blood before it returns to the
    # lung, which sets the mixed-venous content
    mixed_venous_content = calcMixedVenousContent(
        end_capillary_content_ml_dl = end_capillary$total_ml_dl,
        vo2_ml_min = vo2_ml_min,
        shunt_fraction = shunt_fraction,
        cardiac_output_l_min = cardiac_output_l_min
    )

    # Shunted mixed-venous blood then mixes with end-capillary blood by content
    arterial_content = calcArterialContent(
        end_capillary_content_ml_dl = end_capillary$total_ml_dl,
        mixed_venous_content_ml_dl = mixed_venous_content,
        shunt_fraction = shunt_fraction
    )

    # Convert the calculated contents back to PO2 for display
    arterial_po2 = calcPO2fromContent(
        content_ml_dl = arterial_content,
        hb_g_l = hb_g_l,
        pco2_mmhg = pco2_mmhg,
        plasma_ph = plasma_ph,
        dpg_mmol_l = dpg_mmol_l,
        temperature_c = temperature_c
    )

    mixed_venous_po2 = calcPO2fromContent(
        content_ml_dl = mixed_venous_content,
        hb_g_l = hb_g_l,
        pco2_mmhg = pco2_mmhg,
        plasma_ph = plasma_ph,
        dpg_mmol_l = dpg_mmol_l,
        temperature_c = temperature_c
    )

    arterial = calcOxygenContent(
        po2_mmhg = arterial_po2,
        hb_g_l = hb_g_l,
        pco2_mmhg = pco2_mmhg,
        plasma_ph = plasma_ph,
        dpg_mmol_l = dpg_mmol_l,
        temperature_c = temperature_c
    )
    mixed_venous = calcOxygenContent(
        po2_mmhg = mixed_venous_po2,
        hb_g_l = hb_g_l,
        pco2_mmhg = pco2_mmhg,
        plasma_ph = plasma_ph,
        dpg_mmol_l = dpg_mmol_l,
        temperature_c = temperature_c
    )

    # Combine the three blood states in display order
    rows = rbind(end_capillary, mixed_venous, arterial)
    rows$state = c("end_capillary", "mixed_venous", "arterial")
    rows$label = c(
        "Pulmonary end-capillary",
        "Mixed venous",
        "Arterial"
    )
    rows$short_label = c("End-capillary", "Mixed venous", "Arterial")
    rows$content_ml_dl = c(
        end_capillary$total_ml_dl,
        mixed_venous_content,
        arterial_content
    )

    rows[,
        c(
            "state",
            "label",
            "short_label",
            "po2_mmhg",
            "saturation",
            "bound_ml_dl",
            "dissolved_ml_dl",
            "content_ml_dl"
        )
    ]
}

plot_state_data = function(state, outcome) {
    plotData = state

    # Aesthetics
    ## Colours
    plotData$colour = bloodColour(plotData$saturation)

    ## Label position
    label_direction = c(
        end_capillary = 1,
        mixed_venous = -1,
        arterial = 1
    )[plotData$state]

    label_direction[plotData$po2_mmhg < 120] = 1
    label_direction[plotData$po2_mmhg > 615] = -1

    plotData$label_x = plotData$po2_mmhg + 18 * label_direction
    plotData$label_hjust = ifelse(label_direction > 0, 0, 1)

    # Values
    ## SpO2
    if (outcome == "saturation") {
        plotData$value = 100 * plotData$saturation

        label_offset = c(
            end_capillary = 0,
            mixed_venous = -4,
            arterial = -8
        )[plotData$state]

        plotData$label_value = plotData$value + label_offset

        label_ceiling = c(
            end_capillary = 97.5,
            mixed_venous = 92.5,
            arterial = 87.5
        )[plotData$state]

        plotData$label_value = pmin(
            label_ceiling,
            pmax(2.5, plotData$label_value)
        )
        ## Content
    } else {
        plotData$value = plotData$content_ml_dl
        plotData$label_value = plotData$value + c(1.2, 0, -1.2)
        plotData$label_value = pmax(0.5, plotData$label_value)
    }

    plotData
}

# Aesthetics
# Function to calculate the label/dot colours based on the oxygen saturation
bloodColour = function(saturation) {
    saturation = pmin(1, pmax(0, saturation))
    saturation_anchors = c(0, 0.70, 0.80, 0.90, 0.95, 0.98, 1)
    colour_anchors = c(
        "#17324D",
        "#2D6F9E",
        "#347EAE",
        "#735678",
        "#B83240",
        "#E45B58",
        "#F06A5F"
    )
    anchor_channels = t(grDevices::col2rgb(colour_anchors))
    channels = matrix(
        numeric(length(saturation) * ncol(anchor_channels)),
        nrow = length(saturation),
        ncol = ncol(anchor_channels)
    )

    for (channel in seq_len(ncol(anchor_channels))) {
        channels[, channel] = stats::approx(
            x = saturation_anchors,
            y = anchor_channels[, channel],
            xout = saturation,
            rule = 2
        )$y
    }

    grDevices::rgb(
        channels[, 1],
        channels[, 2],
        channels[, 3],
        maxColorValue = 255
    )
}

## Format one calculated blood state for display in the diagram
shuntDiagramState = function(state, state_name) {
    selected_state = state[
        state$state == state_name,
        ,
        drop = FALSE
    ]

    if (nrow(selected_state) != 1) {
        stop("state must contain each blood state once", call. = FALSE)
    }

    list(
        po2 = sprintf("%.1f", selected_state$po2_mmhg),
        saturation = sprintf("%.1f%%", 100 * selected_state$saturation),
        content = sprintf("%.2f", selected_state$content_ml_dl),
        colour = unname(bloodColour(selected_state$saturation))
    )
}

STANDARD_P50_MMHG = 26.8

## Calculate the p50 of the oxyHb curve based on CO2, pH, 2-3 DPG, and temperature
calculateP50 = function(
    pco2_mmhg = STANDARD_PCO2_MMHG,
    plasma_ph = 7.40,
    dpg_mmol_l = 4.65,
    temperature_c = STANDARD_TEMPERATURE_C
) {
    validateScalar(pco2_mmhg, "pco2_mmhg", 0, 200)
    validateScalar(plasma_ph, "plasma_ph", 6.5, 8.0)
    validateScalar(dpg_mmol_l, "dpg_mmol_l", 0, 15)
    validateScalar(temperature_c, "temperature_c", 10, 45)

    rbc_ph = plasma_ph + log10(RBC_DONNAN_RATIO)
    dpg_mol_l = dpg_mmol_l / 1000

    delta_rbc_ph = rbc_ph - STANDARD_RBC_PH
    delta_pco2 = pco2_mmhg - STANDARD_PCO2_MMHG
    delta_dpg = dpg_mol_l - STANDARD_DPG_MOL_L
    delta_temperature = temperature_c - STANDARD_TEMPERATURE_C

    # Official MML section P50_adj (lines 121-130):
    # P501-P504 are the pH, PCO2, DPG, and temperature adjustments
    # The p50 is their multiplicative combination; relative to the standard P50
    #fmt: skip
    p50_ph = STANDARD_P50_MMHG +
        1.2 * (-21.279 * delta_rbc_ph +
               8.872   * delta_rbc_ph^2 -
               1.47    * delta_rbc_ph^3
        )

    #fmt: skip
    p50_co2 = STANDARD_P50_MMHG +
        1.7 * (4.28e-2 * delta_pco2 +
               3.64e-5 * delta_pco2^2
        )

    #fmt: skip
    p50_dpg = STANDARD_P50_MMHG +
        (795.633533   * delta_dpg
         - 19660.8947 * delta_dpg^2
        )

    #fmt: skip
    p50_temperature = STANDARD_P50_MMHG +
        0.98 * (1.4945   * delta_temperature +
                4.335e-2 * delta_temperature^2 +
                7e-4     * delta_temperature^3
        )

    #fmt: skip
    p50 = STANDARD_P50_MMHG *
        (p50_ph / STANDARD_P50_MMHG) *
        (p50_co2 / STANDARD_P50_MMHG) *
        (p50_dpg / STANDARD_P50_MMHG) *
        (p50_temperature / STANDARD_P50_MMHG)

    if (!is.finite(p50) || p50 <= 0) {
        stop("Calculated P50 must be positive", call. = FALSE)
    }

    p50
}

## Calculate SpO2 based on the gas partial pressures and the p50
calculateSpO2 = function(
    po2_mmhg,
    pco2_mmhg = STANDARD_PCO2_MMHG,
    plasma_ph = 7.40,
    dpg_mmol_l = 4.65,
    temperature_c = STANDARD_TEMPERATURE_C
) {
    validateNumeric(po2_mmhg, "po2_mmhg", 0, 1000)

    p50 = calculateP50(
        pco2_mmhg = pco2_mmhg,
        plasma_ph = plasma_ph,
        dpg_mmol_l = dpg_mmol_l,
        temperature_c = temperature_c
    )

    # Official MML sections HillCoeff and HbO2Saturate (lines 143-161):
    # nH = alpha - beta * 10^(-PO2/gamma)
    # Substitution of the equilibrium terms reduces SHbO2 to this variable-Hill relationship
    hill_coefficient = 2.8 - 1.2 * 10^(-po2_mmhg / 29.2)
    pressure_ratio = po2_mmhg / p50
    ratio_power = pressure_ratio^hill_coefficient

    ratio_power / (1 + ratio_power)
}

axisProjectionScale = function(
    plotted_state,
    major_breaks,
    digits,
    min_separation,
    label_separation = 0,
    label_priority = seq_len(nrow(plotted_state))
) {
    # Replace nearby standard ticks so the calculated values remain readable
    major_breaks = major_breaks[vapply(
        major_breaks,
        function(value) {
            all(abs(value - plotted_state$value) >= min_separation)
        },
        logical(1)
    )]
    breaks = sort(unique(c(major_breaks, plotted_state$value)))
    labels = vapply(
        breaks,
        function(value) format(value, trim = TRUE, scientific = FALSE),
        character(1)
    )

    # Keep the highest-priority label when calculated ticks are too close
    visible_labels = rep(TRUE, nrow(plotted_state))
    retained_labels = integer()
    priority_order = order(label_priority, decreasing = TRUE)

    for (state_index in priority_order) {
        overlaps_retained = any(
            abs(
                plotted_state$value[state_index] -
                    plotted_state$value[retained_labels]
            ) < label_separation
        )

        if (overlaps_retained) {
            visible_labels[state_index] = FALSE
        } else {
            retained_labels = c(retained_labels, state_index)
        }
    }

    # Colour only the calculated tick labels
    for (state_index in seq_len(nrow(plotted_state))) {
        break_index = which.min(abs(breaks - plotted_state$value[state_index]))
        labels[break_index] = ""

        if (!visible_labels[state_index]) next

        label_format = paste0(
            "<span style='color: ",
            plotted_state$colour[state_index],
            "'>%.",
            digits,
            "f</span>"
        )
        labels[break_index] = sprintf(
            label_format,
            plotted_state$value[state_index]
        )
    }

    list(breaks = breaks, labels = labels)
}

server = function(input, output, session) {
    p50_parameters = reactive({
        req(
            input$pco2_mmhg,
            input$plasma_ph,
            input$dpg_mmol_l,
            input$temperature_c
        )

        list(
            pco2_mmhg = input$pco2_mmhg,
            plasma_ph = input$plasma_ph,
            dpg_mmol_l = input$dpg_mmol_l,
            temperature_c = input$temperature_c
        )
    })

    # The consumption the slider offers, being whichever of the app's range and
    # oxygen delivery binds first. Truncated to the whole mL/min the slider
    # steps in so the value it settles on is always deliverable.
    # Deliberately independent of input$vo2: taking a dependency on it would
    # re-fire the observer below on every drag of the consumption slider.
    vo2_maximum = reactive({
        req(
            input$end_capillary_po2,
            input$cardiac_output,
            input$shunt_fraction,
            input$hb_g_l
        )

        parameters = p50_parameters()
        end_capillary_content = calcOxygenContent(
            po2_mmhg = input$end_capillary_po2,
            hb_g_l = input$hb_g_l,
            pco2_mmhg = parameters$pco2_mmhg,
            plasma_ph = parameters$plasma_ph,
            dpg_mmol_l = parameters$dpg_mmol_l,
            temperature_c = parameters$temperature_c
        )$total_ml_dl

        min(
            MAX_VO2_ML_MIN,
            floor(calcMaxVO2(
                end_capillary_content_ml_dl = end_capillary_content,
                shunt_fraction = input$shunt_fraction / 100,
                cardiac_output_l_min = input$cardiac_output
            ))
        )
    })

    # The slider's range follows the circulation, so a consumption the tissues
    # could never be supplied cannot be asked for. Only a change in delivery
    # moves the range, which leaves a drag on the consumption slider alone.
    observeEvent(vo2_maximum(), {
        updateSliderInput(
            session,
            "vo2",
            max = vo2_maximum(),
            value = min(input$vo2, vo2_maximum())
        )
    })

    # The narrowed slider only reports back after a round trip to the browser,
    # so the model runs against the deliverable consumption in the meantime
    modelled_vo2 = reactive({
        req(input$vo2)

        min(input$vo2, vo2_maximum())
    })

    model_result = reactive({
        parameters = p50_parameters()
        tryCatch(
            list(
                value = calcShuntState(
                    end_capillary_po2_mmhg = input$end_capillary_po2,
                    shunt_fraction = input$shunt_fraction / 100,
                    vo2_ml_min = modelled_vo2(),
                    cardiac_output_l_min = input$cardiac_output,
                    hb_g_l = input$hb_g_l,
                    pco2_mmhg = parameters$pco2_mmhg,
                    plasma_ph = parameters$plasma_ph,
                    dpg_mmol_l = parameters$dpg_mmol_l,
                    temperature_c = parameters$temperature_c
                ),
                error = NULL
            ),
            error = function(error) {
                list(value = NULL, error = conditionMessage(error))
            }
        )
    })

    model_state = reactive({
        result = model_result()
        validate(need(is.null(result$error), result$error))
        result$value
    })

    ## Read one blood state's oxygen content out of the calculated model
    stateContent = function(state, state_name) {
        state$content_ml_dl[state$state == state_name]
    }

    # Extraction is reported from the blood the model has already calculated
    output$extraction_ratio = renderText({
        state = model_state()
        extraction_ratio = calcExtractionRatio(
            arterial_content_ml_dl = stateContent(state, "arterial"),
            mixed_venous_content_ml_dl = stateContent(state, "mixed_venous")
        )

        sprintf("%.1f%%", 100 * extraction_ratio)
    })

    observe({
        session$sendCustomMessage(
            "update-shunt-diagram",
            buildShuntDiagramData(
                state = model_state(),
                shunt_fraction = input$shunt_fraction / 100
            )
        )
    })

    current_p50 = reactive({
        parameters = p50_parameters()

        calculateP50(

(function (global) {
  "use strict";

  function roundCoordinate(value) {
    return Number(value.toFixed(2));
  }

  function tubePath(points, width) {
    const halfWidth = width / 2;
    const leftEdge = [];
    const rightEdge = [];

    points.forEach((point, index) => {
      const previous = points[Math.max(0, index - 1)];
      const next = points[Math.min(points.length - 1, index + 1)];
      const dx = next.x - previous.x;
      const dy = next.y - previous.y;
      const length = Math.hypot(dx, dy) || 1;
      const normalX = -dy / length;
      const normalY = dx / length;

      leftEdge.push({
        x: point.x + normalX * halfWidth,
        y: point.y + normalY * halfWidth
      });
      rightEdge.push({
        x: point.x - normalX * halfWidth,
        y: point.y - normalY * halfWidth
      });
    });

    return leftEdge.concat(rightEdge.reverse()).map((point, index) =>
      `${index === 0 ? "M" : "L"}${roundCoordinate(point.x)} ${roundCoordinate(point.y)}`
    ).join(" ") + " Z";
  }

  function variableTubePath(points, widths) {
    if (points.length !== widths.length) {
      throw new Error("Each vessel point requires a corresponding width");
    }

    const leftEdge = [];
    const rightEdge = [];
    points.forEach((point, index) => {
      const previous = points[Math.max(0, index - 1)];
      const next = points[Math.min(points.length - 1, index + 1)];
      const dx = next.x - previous.x;
      const dy = next.y - previous.y;
      const length = Math.hypot(dx, dy) || 1;
      const halfWidth = widths[index] / 2;
      const normalX = -dy / length;
      const normalY = dx / length;
      leftEdge.push({ x: point.x + normalX * halfWidth, y: point.y + normalY * halfWidth });
      rightEdge.push({ x: point.x - normalX * halfWidth, y: point.y - normalY * halfWidth });
    });

    return leftEdge.concat(rightEdge.reverse()).map((point, index) =>
      `${index === 0 ? "M" : "L"}${roundCoordinate(point.x)} ${roundCoordinate(point.y)}`
    ).join(" ") + " Z";
  }

  function hourglassWidths(points, junctionWidth, middleWidth) {
    const lengths = points.slice(1).map((point, index) =>
      Math.hypot(point.x - points[index].x, point.y - points[index].y)
    );
    const totalLength = lengths.reduce((sum, length) => sum + length, 0);
    let distance = 0;
    return points.map((point, index) => {
      if (index > 0) distance += lengths[index - 1];
      const fraction = totalLength === 0 ? 0 : distance / totalLength;
      const narrowing = Math.sin(Math.PI * fraction) ** 2;
      return junctionWidth - (junctionWidth - middleWidth) * narrowing;
    });
  }

  function sampleCubic(start, firstControl, secondControl, end, steps = 24) {
    return Array.from({ length: steps + 1 }, (_, index) => {
      const t = index / steps;
      const inverse = 1 - t;
      return {
        x: inverse ** 3 * start.x +
          3 * inverse ** 2 * t * firstControl.x +
          3 * inverse * t ** 2 * secondControl.x + t ** 3 * end.x,
        y: inverse ** 3 * start.y +
          3 * inverse ** 2 * t * firstControl.y +
          3 * inverse * t ** 2 * secondControl.y + t ** 3 * end.y
      };
    });
  }

  function symmetricCubicRoute(start, end, sag, steps = 48) {
    const offset = (end.x - start.x) * 0.28;
    return sampleCubic(
      start,
      { x: start.x + offset, y: start.y + sag },
      { x: end.x - offset, y: end.y + sag },
      end,
      steps
    );
  }

  function circlePath(point, diameter) {
    const radius = diameter / 2;
    const left = roundCoordinate(point.x - radius);
    const right = roundCoordinate(point.x + radius);
    const y = roundCoordinate(point.y);
    const r = roundCoordinate(radius);
    return `M${right} ${y} A${r} ${r} 0 1 0 ${left} ${y} A${r} ${r} 0 1 0 ${right} ${y} Z`;
  }

  function positionArrow(root, selector, points, fraction) {
    const lengths = points.slice(1).map((point, index) =>
      Math.hypot(point.x - points[index].x, point.y - points[index].y)
    );
    const totalLength = lengths.reduce((sum, length) => sum + length, 0);
    let target = totalLength * Math.max(0, Math.min(1, fraction));
    let segmentIndex = 0;
    while (segmentIndex < lengths.length - 1 && target > lengths[segmentIndex]) {
      target -= lengths[segmentIndex];
      segmentIndex += 1;
    }
    const start = points[segmentIndex];
    const end = points[segmentIndex + 1];
    const weight = target / (lengths[segmentIndex] || 1);
    const arrow = root.querySelector(selector);
    arrow.style.left = `${roundCoordinate(start.x + (end.x - start.x) * weight)}px`;
    arrow.style.top = `${roundCoordinate(start.y + (end.y - start.y) * weight)}px`;
    arrow.style.setProperty(
      "--arrow-angle",
      `${roundCoordinate(Math.atan2(end.y - start.y, end.x - start.x) * 180 / Math.PI)}deg`
    );
  }

  function setClipPath(root, selector, parts) {
    const node = root.querySelector(selector);
    const path = parts.length > 0 ? `path("${parts.join(" ")}")` : "inset(0 0 100% 0)";
    node.style.clipPath = path;
    node.style.webkitClipPath = path;
  }

  function updateVascularGeometry(root, shuntFraction) {
    const parentWidth = 24;
    const wallAddition = 6;
    const shuntWidth = parentWidth * Math.sqrt(shuntFraction);
    const ventilatedWidth = parentWidth * Math.sqrt(1 - shuntFraction);
    const split = { x: 270, y: 410 };
    const mixer = { x: 800, y: 410 };
    const capillaryInlet = { x: 410, y: 360 };
    const capillaryOutlet = { x: 670, y: 360 };
    const ventilatedInlet = sampleCubic(
      split, { x: 320, y: 410 }, { x: 370, y: 360 }, capillaryInlet, 32
    );
    const capillary = sampleCubic(
      capillaryInlet, { x: 480, y: 360 }, { x: 600, y: 360 }, capillaryOutlet, 64
    );
    const endcapOutlet = sampleCubic(
      capillaryOutlet, { x: 710, y: 360 }, { x: 760, y: 410 }, mixer, 32
    );
    const ventilatedRoute = ventilatedInlet.slice(0, -1)
      .concat(capillary.slice(0, -1), endcapOutlet);
    const inlet = [{ x: 55, y: 410 }, split];
    const shunt = symmetricCubicRoute(split, mixer, 20, 64);
    const outlet = [mixer, { x: 950, y: 410 }];
    const wallParentWidth = parentWidth + wallAddition;
    const ventilatedJunctionWidth = Math.min(parentWidth, ventilatedWidth * 1.08);
    const shuntJunctionWidth = shuntWidth * 1.12;
    const ventilatedWidths = hourglassWidths(
      ventilatedRoute, ventilatedJunctionWidth, ventilatedWidth
    );
    const shuntWidths = hourglassWidths(shunt, shuntJunctionWidth, shuntWidth);
    const wallVentilatedWidths = ventilatedWidths.map((width) => width + wallAddition);
    const wallShuntWidths = shuntWidths.map((width) => width + wallAddition);

    const wallParts = [
      tubePath(inlet, wallParentWidth),
      variableTubePath(ventilatedRoute, wallVentilatedWidths),
      tubePath(outlet, wallParentWidth),
      circlePath(inlet[0], wallParentWidth),
      circlePath(split, wallParentWidth),
      circlePath(mixer, wallParentWidth),
      circlePath(outlet[1], wallParentWidth),
      shuntFraction === 0 ? tubePath(shunt, 2) : variableTubePath(shunt, wallShuntWidths)
    ];
    const parentParts = [
      tubePath(inlet, parentWidth), circlePath(inlet[0], parentWidth),
      circlePath(split, parentWidth)
    ];
    const shuntParts = shuntFraction > 0
      ? [variableTubePath(shunt, shuntWidths)]
      : [];
    const arterialParts = [
      tubePath(outlet, parentWidth), circlePath(mixer, parentWidth),
      circlePath(outlet[1], parentWidth)
    ];

    setClipPath(root, "[data-vascular-wall]", wallParts);
    setClipPath(root, "[data-vascular-parent]", parentParts);
    setClipPath(root, "[data-vascular-ventilated]", [
      variableTubePath(ventilatedRoute, ventilatedWidths)
    ]);
    setClipPath(root, "[data-vascular-shunt]", shuntParts);
    setClipPath(root, "[data-vascular-arterial]", arterialParts);
    positionArrow(root, ".inlet-flow-arrow", inlet, 0.55);
    positionArrow(root, ".shunt-flow-arrow", shunt, 0.55);
    positionArrow(root, ".ventilated-flow-arrow", ventilatedInlet, 0.55);
    positionArrow(root, ".upper-capillary-flow-arrow", capillary, 0.52);
    positionArrow(root, ".endcap-flow-arrow", endcapOutlet, 0.48);
    positionArrow(root, ".outlet-flow-arrow", outlet, 0.58);
  }

    const trunk = [inlet, bifurcation];

      bifurcation, { x: 420, y: 70 }, { x: 300, y: 96 }, deadSpaceNeck, 40
    );

      bifurcation, { x: 510, y: 78 }, { x: 540, y: 116 }, idealNeck, 32
    );
    const wallParts = [
      tubePath(trunk, wallWidth), tubePath(deadSpaceBranch, wallWidth),
      tubePath(idealBranch, wallWidth), circlePath(inlet, wallWidth),
      circlePath(bifurcation, wallWidth), circlePath(deadSpaceCenter, 175),
      circlePath(idealCenter, 200)
    ];
    const lumenParts = [
      tubePath(trunk, lumenWidth), tubePath(deadSpaceBranch, lumenWidth),
      tubePath(idealBranch, lumenWidth), circlePath(inlet, lumenWidth),
      circlePath(bifurcation, lumenWidth), circlePath(deadSpaceCenter, 169),

    ];
    setClipPath(root, "[data-airway-wall]", wallParts);

  }

  function updateDiagram(message) {
